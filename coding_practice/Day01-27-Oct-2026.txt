## Ansible Task — Install Nginx on Pathnex Server

---
- name : Install nginx server on Pathnex
  hosts: all
  become: true

  tasks:
   - name: Install Nginx
     yum:
      name: nginx
      state: present

##Kubernetes Taks - creating nginx pod 

apiVersion: v1
kind: Pod
metadata:
  name: ng-pod

 spec:
  containers:
    - name: nginx-pod
     image: nginx:latest
     ports:
      - containerPort: 8080
